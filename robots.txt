# NOTE ON STRUCTURE
# A crawler obeys exactly ONE group: the most specific User-agent that matches.
# A named group therefore REPLACES the "*" group rather than adding to it, so
# every protection has to be repeated in every named group. The previous file
# listed the named crawlers with "Allow: /" only, which silently exempted
# Googlebot and the AI crawlers from the unsubscribe/feedback rules below.
# ---------------------------------------------------------------------------

Disallow: /unsubscribe.html
Disallow: /feedback-good.html
Disallow: /feedback-fine.html
Disallow: /feedback-not-good.html
Disallow: /*/unsubscribe.html
Disallow: /*/feedback-
